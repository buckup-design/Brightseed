import { useState } from "react";
import { Meta } from "@storybook/addon-docs/blocks";

<Meta title="Prototyping Workflow" />

export const CopyInstall = () => {
  const command = "npm install -D @storybook/addon-mcp";
  const [copied, setCopied] = useState(false);
  const copy = () => {
    if (navigator.clipboard) {
      navigator.clipboard.writeText(command).then(() => {
        setCopied(true);
        setTimeout(() => setCopied(false), 1500);
      });
    }
  };
  return (
    <button
      type="button"
      onClick={copy}
      title="Copy to clipboard"
      aria-label="Copy install command to clipboard"
      style={{ display: "inline-flex", alignItems: "center", gap: "0.75rem", background: "#f3f4f5", border: "none", borderRadius: 8, padding: "0.7rem 1rem", fontFamily: "'SFMono-Regular', Consolas, monospace", fontSize: 17, color: "#1b1c1d", cursor: "pointer" }}
    >
      {command}
      {copied ? (
        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="#2e7d32" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
          <path d="M20 6 9 17l-5-5" />
        </svg>
      ) : (
        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="#73828c" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">
          <rect x="9" y="9" width="13" height="13" rx="2" />
          <path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1" />
        </svg>
      )}
    </button>
  );
};

# Prototyping Workflow

## A few conventions worth knowing

- **Light-first.** Build for light; dark comes free via the token override. Never write dark-specific styles.
- **Tag colors are decorative**, not status. Use them to tell items apart; for status, compose icon + text.
- **Icons come from the approved set** (Foundations → Icons). Not there → flag it, don't draw a new one.
- **Data series** use the `--chart-*` colors, never brand lime or forest.

When a written doc ever disagrees with the live tokens or Storybook, the running system wins, and the disagreement is worth flagging so the docs get fixed.
